struct ComboHit {
    string anim;
    float damage;
    float forwardStep;
    float hitPause;
};

ComboHit KH1Combo[3] = {
    { "kh1_sora_attack1.anim", 32.0f, 0.45f, 0.06f },
    { "kh1_sora_attack2.anim", 28.0f, 0.50f, 0.07f },
    { "kh1_sora_attack3.anim", 44.0f, 0.70f, 0.10f }
};

int comboIndex = 0;
float comboTimer = 0;

void UpdateCombo() {
    if (Input.Attack && comboTimer <= 0) {
        PlayAnimation(KH1Combo[comboIndex].anim);
        DealDamage(KH1Combo[comboIndex].damage);
        MoveForward(KH1Combo[comboIndex].forwardStep);
        PauseHit(KH1Combo[comboIndex].hitPause);

        comboIndex++;
        if (comboIndex > 2) comboIndex = 0;

        comboTimer = 0.32f;
    }

    comboTimer -= dt;
}
