float RUN_SPEED = 6.2f;
float ACCEL = 14.0f;
float DECEL = 18.0f;

float JUMP_FORCE = 7.8f;
float JUMP_TIME = 0.42f;
float GRAVITY = 28.0f;

float ROLL_DISTANCE = 3.4f;
float ROLL_IFRAMES = 0.18f;
float ROLL_SPEED = 9.0f;

void UpdateMovement() {
    if (Input.Run) {
        velocity = Lerp(velocity, RUN_SPEED, ACCEL * dt);
    } else {
        velocity = Lerp(velocity, 0, DECEL * dt);
    }

    if (Input.Jump && IsGrounded()) {
        StartJump(JUMP_FORCE, JUMP_TIME);
    }

    if (Input.Roll && CanRoll()) {
        StartRoll(ROLL_SPEED, ROLL_DISTANCE, ROLL_IFRAMES);
    }
}
